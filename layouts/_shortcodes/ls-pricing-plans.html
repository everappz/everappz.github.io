{{- /* pricing-plans: render an app's pricing plans from data/pricing.yaml. App key comes from the page directory (or a `product` param). Each feature shows title (left) + limit (right): number -> red text | X -> red cross | V -> tint check | U -> tint ∞ Grid columns follow the number of plans. @example {{< pricing-plans >}} */ -}} {{- $app := .Get "product" | default (path.Base .Page.File.Dir) -}} {{- $data := index hugo.Data.pricing $app -}} {{- with $data -}} {{- $features := .features -}} {{- $cols := len .plans -}} {{- /* App Store link built from the single source of truth (data/apps.yaml). */ -}} {{- $iosId := index (index hugo.Data.apps $app) "ios" -}} {{- $storeLink := printf "https://apps.apple.com/app/apple-store/id%s?%s" $iosId site.Params.appStoreAffiliate -}} {{- $iconCross := (resources.Get "icons/pricing/cross.svg").Content -}} {{- $iconCheck := (resources.Get "icons/pricing/check.svg").Content -}} {{- $iconInfinity := (resources.Get "icons/pricing/infinity.svg").Content -}}
{{- range $plan := .plans -}}

{{ i18n $plan.titleKey }}

{{- range $i, $ft := $features -}} {{- $limit := index $plan.limits $i -}}
{{ i18n $ft }} {{- if findRE `^[0-9]` $limit -}} {{- $disp := $limit -}} {{- if hasSuffix $limit "/day" -}}{{- $disp = printf "%s%s" (strings.TrimSuffix "/day" $limit) (i18n "pricingRatePerDay") -}}{{- end -}} {{ $disp }} {{- else if eq $limit "X" -}} {{ $iconCross | safeHTML }} {{- else if eq $limit "V" -}} {{ $iconCheck | safeHTML }} {{- else if eq $limit "U" -}} {{ $iconInfinity | safeHTML }} {{- end -}}
{{- end -}}
{{- with $plan.button -}} {{- /* Paid plans show the bare price; free shows its localized label. */ -}} {{- $label := i18n .key -}} {{- with .price -}}{{- $label = . -}}{{- end -}} {{ $label }} {{- end -}}
{{- end -}}
{{- else -}} {{- warnf "pricing-plans: no data/pricing entry for %q" $app -}} {{- end -}}
