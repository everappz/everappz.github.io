{{- /* Renders one or more official Apple App Store / Mac App Store badges, plus an async subtitle (version • update date • min iOS • min macOS) fetched client-side from the iTunes Lookup API. Pass "product:platform" pairs (IDs resolved from data/apps.yaml). platform is one of: ios (default), macos/mac, pro. {{< app-store-badges products="evermusic:ios, evermusic:macos" >}} With no products, it falls back to the current page's product (iOS + Mac). */ -}} {{- $aff := site.Params.appStoreAffiliate -}} {{- $pairs := slice -}} {{- with .Get "products" -}} {{- range split (replaceRE `\s*,\s*` "," (trim . " ,")) "," -}} {{- if . -}}{{- $pairs = $pairs | append . -}}{{- end -}} {{- end -}} {{- else -}} {{- $key := path.Base $.Page.File.Dir -}} {{- $pairs = slice (printf "%s:ios" $key) (printf "%s:macos" $key) -}} {{- end -}} {{- $subIos := "" -}} {{- $subMac := "" -}} {{- if not (.Page.Store.Get "appStoreSubtitleAssets") -}} {{- .Page.Store.Set "appStoreSubtitleAssets" true -}} {{- end -}}
{{- range $pairs -}} {{- $parts := split . ":" -}} {{- $prod := trim (index $parts 0) " " -}} {{- $plat := cond (gt (len $parts) 1) (trim (index $parts 1) " ") "ios" -}} {{- with index hugo.Data.apps $prod -}} {{- $id := "" -}}{{- $badgePlat := "ios" -}} {{- if or (eq $plat "macos") (eq $plat "mac") -}}{{- $id = .mac -}}{{- $badgePlat = "macos" -}} {{- else if eq $plat "pro" -}}{{- $id = .pro -}} {{- else -}}{{- $id = .ios -}} {{- end -}} {{- if eq $plat "ios" -}}{{- $subIos = $id -}}{{- end -}} {{- if or (eq $plat "macos") (eq $plat "mac") -}}{{- $subMac = $id -}}{{- end -}} {{- with $id -}} {{- partial "app-store-badge.html" (dict "platform" $badgePlat "link" (printf "https://apps.apple.com/app/apple-store/id%s?%s" . $aff) "page" $.Page) -}} {{- end -}} {{- else -}} {{- warnf "app-store-badges: no data/apps entry for %q" $prod -}} {{- end -}} {{- end -}}
